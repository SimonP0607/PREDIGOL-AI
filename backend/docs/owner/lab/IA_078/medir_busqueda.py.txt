def main(dsn: str, rounds: int = 5, threshold: float = 0.6) -> None:
    conn = psycopg.connect(dsn)
    hits, rows_out, times = 0, [], []
    for q, group, expect, cat in BANK:
        per = []
        for _ in range(rounds):
            t0 = time.perf_counter()
            key = r.search_key(conn, q) or ""
            found = r.entities(conn, key, 8, threshold) if key else []
            teams = [x["entity_id"] for x in found if x["kind"] in ("team", "national_team")][:8]
            r.fixtures(conn, teams, 8)
            if not found and key:
                r.suggestions(conn, key)
            per.append((time.perf_counter() - t0) * 1000)
            conn.rollback()
        ms = statistics.median(per)
        times.append(ms)
        mine = [x for x in found if x["kind"] == group]
        pos = next((i + 1 for i, x in enumerate(mine) if expect.lower() in (x["label"] + " " + x["matched_name"]).lower()), None)
        hits += pos is not None
        rows_out.append((q, group, cat, pos, round(ms, 1), mine[0]["label"] if mine else "-"))
    for row in rows_out:
        print("|".join(str(c) for c in row))
    by_cat: dict[str, list[int]] = {}
    for q, g, cat, pos, ms, top in rows_out:
        by_cat.setdefault(cat, []).append(pos is not None)
    print("recall@8", hits, "/", len(BANK), round(hits / len(BANK), 3))
    for cat, v in sorted(by_cat.items()):
        print("  ", cat, sum(v), "/", len(v))
    times.sort()
    print("latency ms: median", round(statistics.median(times), 1), "p90", round(times[int(0.9 * len(times)) - 1], 1),
          "max", round(max(times), 1), "n", len(times), "rounds", rounds)

if __name__ == "__main__":
    main(sys.argv[1], threshold=float(sys.argv[2]) if len(sys.argv) > 2 else 0.6)
